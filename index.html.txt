const useScrollReveal = () => {
  const [isVisible, setIsVisible] = useState(false);
  const domRef = useRef(null);

  useEffect(() => {
    const observer = new IntersectionObserver(entries => {
      entries.forEach(entry => {
        if (entry.isIntersecting) {
          setIsVisible(true);
        }
      });
    }, { threshold: 0.1, rootMargin: '0px 0px -50px 0px' });

    const currentRef = domRef.current;
    if (currentRef) observer.observe(currentRef);
    return () => {
      if (currentRef) observer.unobserve(currentRef);
    };
  }, []);

  return [domRef, isVisible];
};

const ChatWidget = () => {
  const { t, lang, isChatOpen, setIsChatOpen } = useContext(LanguageContext);
  const [messages, setMessages] = useState([]);
  const [input, setInput] = useState('');
  const messagesEndRef = useRef(null);

  useEffect(() => {
    if (isChatOpen && messages.length === 0) {
      setMessages([{ text: t.chat.initialMsg, isBot: true }]);
    }
  }, [isChatOpen, messages.length, t.chat.initialMsg]);

  useEffect(() => {
    messagesEndRef.current?.scrollIntoView({ behavior: 'smooth' });
  }, [messages]);

  const handleSend = (e) => {
    e.preventDefault();
    if (!input.trim()) return;
    
    const userMsg = input;
    setMessages(prev => [...prev, { text: userMsg, isBot: false }]);
    setInput('');

    setTimeout(() => {
      setMessages(prev => [...prev, { text: t.chat.botReply, isBot: true }]);
    }, 1200);
  };

  if (!isChatOpen) return null;

  return (
    <div className="fixed bottom-6 right-6 z-[100] w-[350px] max-w-[calc(100vw-3rem)] bg-white/90 backdrop-blur-2xl border border-white shadow-[0_20px_50px_rgb(39,76,119,0.15)] rounded-3xl overflow-hidden flex flex-col origin-bottom-right animate-[chatPop_0.3s_ease-out_forwards]">
      <div className="bg-[#274C77] p-4 flex justify-between items-center text-white shadow-md z-10">
        <div className="flex items-center gap-2 font-bold">
          <MessageSquare className="w-5 h-5" />
          {t.chat.title}
        </div>
        <button 
          onClick={() => setIsChatOpen(false)} 
          className="hover:bg-white/20 p-1.5 rounded-full transition-colors active:scale-95"
        >
          <X className="w-5 h-5"/>
        </button>
      </div>
      
      <div className="h-80 p-4 overflow-y-auto flex flex-col gap-3 bg-[#E7ECEF]/30">
        {messages.map((msg, idx) => (
          <div key={idx} className={`max-w-[85%] p-3.5 rounded-2xl text-sm font-semibold shadow-sm ${msg.isBot ? 'bg-white border border-white text-[#274C77] self-start rounded-tl-sm' : 'bg-[#274C77] text-white self-end rounded-tr-sm'}`}>
            {msg.text}
          </div>
        ))}
        <div ref={messagesEndRef} />
      </div>
      
      <form onSubmit={handleSend} className="p-3 bg-white border-t border-[#E7ECEF] flex gap-2">
        <input 
          type="text" 
          value={input}
          onChange={(e) => setInput(e.target.value)}
          placeholder={t.chat.placeholder}

export default function App() {
  const [lang, setLang] = useState('en');
  const [isChatOpen, setIsChatOpen] = useState(false);
  const t = translations[lang];

  const scrollToSection = (sectionId) => {
    const section = document.getElementById(sectionId);
    if (section) {
      section.scrollIntoView({ behavior: 'smooth' });
    }
  };

  return (
    <LanguageContext.Provider value={{ lang, setLang, t, isChatOpen, setIsChatOpen, scrollToSection }}>

        <ChatWidget />

}
